# The Brimstone Sanctum building blocks (docs/structures/brimstone-sanctums.md#building-blocks).
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/brimstone-blocks.txt
# Expect in latest.log:
# - each /loot line drops its own block (2 slabs from the double slab)
# - each /recipe line unlocks 1 recipe ("No new recipes were learned" once an earlier run has unlocked them)
# - using the Contract Lectern prints "The Law of the Brimstone Covenant" and its four laws ([CHAT]) and logs
#   "[ContractLectern] Dev read the Covenant's law at ..."
# - the Pactbroker's trades 9 and 10: `slots` shows 16 Brimstone, then 8 Brimstone Bricks, in the output slot
# Screenshots: brimstone-blocks-display (every block at noon), brimstone-blocks-night (the lights, sigil and
# script at midnight), brimstone-blocks-close (close-up of the materials), brimstone-blocks-lights (close-up of the
# lectern, sconce and censers).

closescreen

/gamemode creative
/gamerule doDaylightCycle false
/time set noon
/fill 18 149 -8 36 149 12 dndclasses:brimstone

/fill 26 150 8 26 151 8 dndclasses:infernal_script_panel[facing=north]
/fill 27 150 8 27 151 8 dndclasses:sulphur_vent
/fill 28 150 8 31 152 8 dndclasses:brimstone_bricks
# Shapes in front
/setblock 20 150 6 dndclasses:brimstone_stairs[facing=south]
/setblock 21 150 6 dndclasses:brimstone_slab[type=bottom]
/setblock 22 150 6 dndclasses:brimstone_wall

/setblock 28 150 6 dndclasses:brimstone_brick_wall
/setblock 29 150 6 dndclasses:cut_hellbrass_stairs[facing=south]
/setblock 30 150 6 dndclasses:cut_hellbrass_slab[type=bottom]
# Sigil tiles on the floor, a vent fuming in it
/fill 20 149 3 22 149 5 dndclasses:pact_sigil_tile
/setblock 25 149 4 dndclasses:sulphur_vent
# The lights and the lectern: a sconce on the brick wall, a censer hung from a beam on a chain, one standing
/setblock 29 151 7 dndclasses:soulfire_sconce[facing=north]
/fill 30 155 5 33 155 7 dndclasses:brimstone_bricks
/setblock 31 154 6 minecraft:chain
/setblock 31 153 6 minecraft:chain
/setblock 31 152 6 dndclasses:chain_hung_censer[hanging=true]
/setblock 33 150 5 dndclasses:chain_hung_censer
/setblock 32 150 4 dndclasses:contract_lectern[facing=north]
/tp @s 26.5 151.5 -2.5 0 18
wait 40

/time set midnight
