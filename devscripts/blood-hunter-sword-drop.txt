# Blood Hunter can't drop a sword (the drop key keeps it in the hotbar), other items and other classes drop normally. Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/blood-hunter-sword-drop.txt
# Expect in latest.log: no iron_sword item entity after the Blood Hunter drop, the dirt drops, and the Fighter's sword drops.
wait 40
closescreen
wait 10
/gamemode survival
/tp @s 0 150 0 0 0
/effect give @s minecraft:slow_falling 60 0 true
/kill @e[type=item]
/clear @s
/dndclass set @s bloodhunter
wait 10
/item replace entity @s hotbar.0 with minecraft:iron_sword
/item replace entity @s hotbar.1 with minecraft:dirt
hotbar 0
wait 5
press key.drop
wait 10
screenshot blood-hunter-sword-drop
/execute if entity @e[type=item,nbt={Item:{id:"minecraft:iron_sword"}}]
/execute if entity @s[nbt={Inventory:[{Slot:0b,id:"minecraft:iron_sword"}]}]
hotbar 1
wait 5
press key.drop
wait 10
/execute if entity @e[type=item,nbt={Item:{id:"minecraft:dirt"}}]
/dndclass set @s fighter
wait 10
hotbar 0
wait 5
press key.drop
wait 10
/execute if entity @e[type=item,nbt={Item:{id:"minecraft:iron_sword"}}]
/kill @e[type=item]
/gamemode creative
quit
