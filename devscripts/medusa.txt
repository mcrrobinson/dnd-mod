# The Medusa (docs/bosses/medusa.md).
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/medusa.txt
# The player is a Wizard on a y=150 platform. Check run/logs/latest.log (and debug.log for [MobAction]):
# 1. Gaze, forced 10 then 2: "[D20] Dev Petrifying Gaze ..." FAILURE (not by 5), "[Conditions] Dev +restrained
#    (Medusa, indefinite, save CON 14/40t -> petrified)", then the repeat save fails: "Dev restrained -> petrified";
#    screenshot medusa-petrified.
# 2. Facing away (forced 2 queued): no new "Petrifying Gaze" line; "rigged rolls" clear count is 1.
# 3. Shield raised at her: "reels from her own reflection!" and "[Conditions] Medusa +stunned (Dev, 20t".
# 4. Statues: NoAI zombies facing her, the first forced 2: "[Conditions] Zombie +petrified (Medusa, indefinite"
#    and its list shows "petrified"; screenshot medusa-statues.
# 5. Fail by 5 (forced 1): straight "[Conditions] Dev +petrified (Medusa, 300t".
# 6. A blind player isn't gazed at; an AI Medusa looses volleys ("[MobAction] Medusa volley -> Dev" in debug.log)
#    and bites when you stick to her ("[MobAction] Medusa bite", "[D20] Dev Snake bite", "+poisoned (Medusa").
# 7. Gorgon's Gaze on a cow (forced 2): "[Magic] Dev used Gorgon's Gaze on Cow: petrified".
/time set noon
/weather clear
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/difficulty normal
wait 40
closescreen
include _reset.txt
respawn
wait 10
/dndclass set @s wizard
wait 5
closescreen
/clear @s
/fill -6 149 -6 14 149 24 minecraft:stone
/fill -6 150 -6 14 156 24 minecraft:air
/kill @e[type=!minecraft:player,distance=..80]
/gamemode survival
/effect clear @s
/effect give @s minecraft:resistance 900 4 true
/effect give @s minecraft:health_boost 900 9 true
/effect give @s minecraft:instant_health 1 9 true
/dndclass forceroll @s clear
/dndcondition remove @s all
wait 5

# ---- 0. Her model, close up (creative: her gaze can't touch you) ----
/gamemode creative
/summon dndclasses:medusa 4.5 150 10.5 {NoAI:1b,Rotation:[180f,0f],Tags:["medtest"],PersistenceRequired:1b}
/tp @s 5.5 151 7.5 20 15
wait 30
clearchat
screenshot medusa-model
/gamemode survival

# ---- 1. Petrifying Gaze: restrained, then a failed repeat save turns you to stone ----
/tp @s 4.5 150 4.5 0 0
/dndclass forceroll @s 10 2
wait 100
/dndcondition list @s
perspective back
wait 5
clearchat
screenshot medusa-petrified
perspective first
/tp @s 4.5 150 4.5 180 0
/dndcondition remove @s all
/dndclass forceroll @s clear
wait 5

# ---- 2. Looking away: no gaze save ----
/tp @s 4.5 150 4.5 180 0
/say step2 facing away
/dndclass forceroll @s 2
wait 120
/dndclass forceroll @s clear
/dndcondition list @s

# ---- 3. A raised shield turns the gaze back on her ----
/say step3 shield
# Raise the shield facing away (no save), then turn to face her behind it.
/item replace entity @s weapon.offhand with minecraft:shield
/tp @s 4.5 150 4.5 180 0
hotbar 8
holduse on
wait 15
look 0 0
wait 60
/dndcondition list @s
holduse off
/item replace entity @s weapon.offhand with minecraft:air
/tp @s 12.5 150 12.5 90 0
/dndcondition remove @s all
/dndclass forceroll @s clear

# ---- 4. Mobs facing her become statues ----
/say step4 statues
/tp @s 12.5 150 12.5 90 0
# Zombies, so the log reads "Zombie +petrified"; noon sun would set them alight, so they wear helmets.
/summon minecraft:zombie 3.5 150 14.5 {NoAI:1b,Rotation:[180f,0f],Tags:["medtest","medzombie"],PersistenceRequired:1b,ArmorItems:[{},{},{},{id:"minecraft:leather_helmet",Count:1b}]}
/summon minecraft:zombie 5.5 150 14.5 {NoAI:1b,Rotation:[180f,0f],Tags:["medtest","medzombie"],PersistenceRequired:1b,ArmorItems:[{},{},{},{id:"minecraft:iron_helmet",Count:1b}]}
/summon minecraft:husk 7.5 150 14.5 {NoAI:1b,Rotation:[150f,0f],Tags:["medtest","medzombie"],PersistenceRequired:1b}
wait 2
/dndclass forceroll @e[type=minecraft:zombie,tag=medzombie,limit=1,sort=nearest] 2
wait 110
/dndcondition list @e[type=minecraft:zombie,tag=medzombie,limit=1,sort=nearest]
/execute as @e[tag=medzombie] run dndcondition list @s
/tp @s 13.5 151 18.5 120 10
wait 10
clearchat
screenshot medusa-statues

# ---- 5. Failing by 5 or more: straight to stone ----
/say step5 fail by five
/dndcondition remove @s all
/tp @s 4.5 150 4.5 0 0
/dndclass forceroll @s 1
wait 120
/dndcondition list @s
/tp @s 4.5 150 4.5 180 0
/dndcondition remove @s all
/dndclass forceroll @s clear

# ---- 6. Blind players are safe; an AI Medusa shoots and bites ----
/say step6 ai medusa
/kill @e[type=dndclasses:medusa]
/effect give @s minecraft:blindness 60 0 true
/tp @s 4.5 150 0.5 0 0
/summon dndclasses:medusa 4.5 150 14.5 {Rotation:[180f,0f],Tags:["medtest"],PersistenceRequired:1b}
wait 100
/dndclass forceroll @s 2
/execute at @e[type=dndclasses:medusa,limit=1] run tp @s ~ ~ ~-1.5
wait 10
/execute at @e[type=dndclasses:medusa,limit=1] run tp @s ~ ~ ~-1.5
wait 10
/execute at @e[type=dndclasses:medusa,limit=1] run tp @s ~ ~ ~-1.5
wait 10
/execute at @e[type=dndclasses:medusa,limit=1] run tp @s ~ ~ ~-1.5
wait 20
/dndcondition list @s
/kill @e[type=dndclasses:medusa]
/kill @e[type=minecraft:arrow]
/effect clear @s minecraft:blindness
/dndcondition remove @s all
/dndclass forceroll @s clear

# ---- 7. Gorgon's Gaze petrifies a cow ----
/say step7 gorgons gaze
/clear @s
/tp @s 4.5 150 4.5 0 20
/summon minecraft:cow 4.5 150 7.5 {NoAI:1b,Rotation:[180f,0f],Tags:["medtest","medcow"],PersistenceRequired:1b}
hotbar 0
/dndmagic release @s all
/dndmagic give @s dndclasses:gorgons_gaze
wait 5
/dndmagic attune @s
wait 5
/dndclass forceroll @e[type=minecraft:cow,tag=medcow,limit=1] 2
use
wait 10
/dndcondition list @e[type=minecraft:cow,tag=medcow,limit=1]
wait 10
clearchat
screenshot medusa-gaze-cow
/kill @e[tag=medtest]
/clear @s
quit
