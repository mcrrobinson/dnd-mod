# Verify PR #215 (mobs skip a DM / Bard / Necromancer / Mob Repel player without dropping their real target).
# A barrier box at 1990..2030 x 149..156 y 2000..2020 z, at night, the player at 2010 150 2010 with Resistance V.
# Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/dm-skip-verify.txt
# Check in latest.log: the "[DevScript] target" lines after each "<phase>" chat line.
#   dm-alone / dm-veiled: every hostile -> none. dm-off-control: they now target Dev.
#   bard-unprovoked: none. bard-after-hit: the zombie Dev hit targets Dev.
#   necro: zombie/skeleton none, spider Dev, Lich Dev.
#   cleric-repel: the zombie that chased Dev drops it; the zombie chasing the villager keeps the villager.
wait 40
closescreen
/spreadplayers 2010 2010 0 1 false @s
include _reset.txt
wait 40
closescreen
/gamemode creative
/dm off
/time set 18000
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/tp @s 2010 160 2010 90 0
wait 40
/kill @e[tag=dms]
/fill 1990 149 2000 2030 156 2020 minecraft:barrier hollow
/fill 1991 150 2001 2029 155 2019 minecraft:air
/tp @s 2010 150 2010 90 0
/gamemode survival
/effect give @s minecraft:resistance 1000 4 true
/effect give @s minecraft:saturation 1000 255 true
/effect give @s minecraft:regeneration 1000 4 true
