# Exercises the event music triggers and the Wyvern boss bar.
# Run: ./gradlew runClient -PdevScript=devscripts/event-music.txt
# Check: run/screenshots/wyvern-boss-bar.png, and "[EventMusic]" lines in run/logs/latest.log
# (expect DRAGON_FIGHT, then NONE, NIGHT, DUNGEON, TRAVEL as each trigger starts and stops).
/time set day
/weather clear
/gamemode creative
/kill @e[type=dndclasses:wyvern]
wait 40
closescreen

# Dragon fight: hurting a wild wyvern as a player shows the boss bar and starts the fight music
/tp @s 12 150 0 90 10
/summon dndclasses:wyvern 0 148 0 {NoAI:1b,NoGravity:1b,Rotation:[0f,0f]}
wait 20
/damage @e[type=dndclasses:wyvern,limit=1] 10 minecraft:player_attack by @s
wait 20
screenshot wyvern-boss-bar
# The fight ends 5 seconds after the last hit (attacker memory)
wait 140
/kill @e[type=dndclasses:wyvern]
wait 20
