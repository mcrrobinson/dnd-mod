# Party roles HUD test (host side, Cleric = Healer): opens the world on LAN port 25631, waits for
# party-roles-guest.txt to join, invites the guest to a party, hurts them and screenshots the party HUD.
# Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/party-roles-host.txt
# then, once the log says "Started serving on": timeout 400 ./gradlew runClient -PdevRunDir=run-2 -PdevServer=localhost:25631 -PdevScript=devscripts/party-roles-guest.txt
wait 40
closescreen
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doWeatherCycle false
/gamerule doMobSpawning false
/gamerule sendCommandFeedback false
/time set 1000
/weather clear
/kill @e[type=!player]
/fill -10 229 -10 20 229 30 minecraft:grass_block
/fill -10 230 -10 20 250 30 minecraft:air
/tp @s 3.5 230 0.5 0 5
/publish false survival 25631
wait 2400
/tp @a[distance=1..] 3.5 230 5.5 180 0
/dndclass set @r[distance=1..] paladin
/dndclass set @s cleric
wait 20
/party invite @r[distance=1..]
wait 200
/gamemode survival @a
/clear @a
/damage @r[distance=1..] 12 minecraft:generic
/give @r[distance=1..] minecraft:diamond_sword
/tp @s 3.5 230 0.5 0 5
wait 20
/party list
hud on
wait 20
screenshot party-roles-host-1
wait 60
screenshot party-roles-host-2
wait 2000
quit
