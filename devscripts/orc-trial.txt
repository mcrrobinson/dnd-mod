# The Trial of Strength: refused for a Human, lost and won without dying as a Half-Orc.
# Run: timeout 600 ./gradlew runClient -PdevScript=devscripts/orc-trial.txt [-PdevWorld=<world with a stronghold>]
# Checks, in <run dir>/logs/latest.log:
# - "[OrcTrial] <player> beat the drum at hold N: refused (not_kin)" while Human
# - "[OrcTrial] <player> started the Trial ..." then "... lost the Trial ... (yielded), health 3.0" after
#   /damage @s 40 (the player is alive at 3 HP), then "refused (tomorrow)" on the next beat
# - after /dndhome trialreset: "started" again, and "... won the Trial ... (champion yielded)" after the
#   Champion takes /damage 60; "Blood and Thunder" in chat (only on the world's first win); the Orcish Greataxe in the slots dump
# - screenshots trial-bar (the duel bar), trial-lost, trial-won
# - each drum "use" logs "use -> block minecraft:note_block": /dndhome stronghold pit aims at the drum's near
#   face (aiming at its middle from the pit floor hit the hay bale under it)
# - no "[DevScript] waitserver: the game is paused" warning (the class picker reopening after /dndrace set used
#   to pause the game, so waitserver never finished)
wait 40
closescreen
include _reset.txt
/gamemode survival
/difficulty normal
/time set 6000
/gamerule doDaylightCycle false
/effect give @s resistance 10 4 true
/dndrace set @s human
# With no class, the race change reopens the class picker, which pauses the game
wait 5
closescreen
/dndhome stronghold pit
waitserver 2
wait 60
use
wait 20
/dndrace set @s halforc
# With no class, the race change reopens the class picker, which pauses the game
wait 5
closescreen
/rep @s dndclasses:bloodtusk_horde set 0
# Forget an earlier run's Trial in this world: a loss today (the clock is stopped) or a win (Blood-Sworn)
/dndhome trialreset @s
/dndhome bloodsworn @s false
# Aim again: the teleport aims from the eye height, which changed with the race
/dndhome stronghold pit
waitserver 2
wait 20
/effect clear @s
/effect give @s instant_health 1 5 true
wait 20
use
wait 40
screenshot trial-bar
/damage @s 40
wait 20
screenshot trial-lost
/data get entity @s Health
# The Champion may have knocked the player off the aim
/dndhome stronghold pit
waitserver 2
wait 40
use
wait 20
/dndhome trialreset @s
/effect give @s instant_health 1 5 true
/dndhome stronghold pit
waitserver 2
wait 40
use
wait 40
/damage @e[type=dndclasses:orc_champion,limit=1,sort=nearest] 60 minecraft:player_attack by @s
wait 20
screenshot trial-won
/data get entity @s Health
slots
wait 40
quit
