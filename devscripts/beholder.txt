# Beholder: model + eyestalk hit shapes, then a live one's anti-magic cone, then its lair.
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/beholder.txt
# Output: run/screenshots/beholder-*.png, and [CHAT] lines in run/logs/latest.log.
/time set day
/weather clear
/gamemode creative
/kill @e[type=dndclasses:beholder]
wait 40
respawn
wait 40
closescreen
wait 20
/effect give @s minecraft:night_vision 999 0 true

# Frozen, facing north towards the camera, high in the sky
/setblock 0 220 -6 minecraft:barrier
/setblock 7 221 0 minecraft:barrier
/tp @s 0.5 221 -5.5 0 5
/summon dndclasses:beholder 0.5 220 0.5 {NoAI:1b,NoGravity:1b,Rotation:[180f,0f]}
wait 40
hud off
screenshot beholder-front
hitboxes on
wait 5
screenshot beholder-hitboxes
hitboxes off
/tp @s 7.5 222 0.5 90 10
wait 20
screenshot beholder-side
hud on
/kill @e[type=dndclasses:beholder]
