# A fresh Dwarven Ruin made with /place structure in new chunks. Shoots the collapsed gatehouse (hall, towers,
# doorway, statues, rubble) from outside, then cuts it away and shoots the layout from overhead.
# 2400, -2400 is about 3400 blocks out: Challenge III (usually).
# Destructive (/dungeon cutaway): use a scratch copy of the run dir.
# Run: timeout 500 ./gradlew runClient -PdevRunDir=run-scratch -PdevScript=devscripts/dungeon-ruin.txt
# If /place says "Failed to place structure", the ground there failed the site check (or the biome isn't a
# mountain or stony shore biome: find one with /locate biome minecraft:jagged_peaks): try another spot.
# Check: screenshots/dungeon-ruin-*.png; the "/dungeon info" [CHAT] lines in logs/latest.log.
wait 40
closescreen
/time set noon
/weather clear
/gamerule doDaylightCycle false
/gamemode creative
/tp @s 2407 200 -2393
wait 200
/place structure dndclasses:dwarven_ruin 2400 64 -2400
wait 100
/dungeon info
# The gatehouse: stand on the ground by the entrance, then back off and look at it
/spreadplayers 2407 -2393 0 1 false @s
wait 40
/tp @s ~-16 ~6 ~-16 -45 20
wait 60
hud off
screenshot dungeon-ruin-gatehouse
hud on
/tp @s ~32 ~ ~ 45 20
wait 40
hud off
screenshot dungeon-ruin-gatehouse-2
hud on
# Overhead, with the dungeon cut away
/gamemode spectator
/dungeon cutaway
wait 60
/tp @s 2407 40 -2393
look 0 90
wait 200
hud off
screenshot dungeon-ruin-overhead
hud on
quit
