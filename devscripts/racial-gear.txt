# Racial armor: set bonus for anyone, race bonus only for the matching race.
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/racial-gear.txt
# Expect in latest.log (ActiveEffects ids: 1 Speed, 3 Haste, 5 Strength, 11 Resistance, 12 Fire Resistance):
#   elven_chain: elf and human both Speed I only; dwarven_plate: dwarf Resistance + Haste, human Resistance only;
#   orcish_warplate after instant damage (under half health): halforc Strength + Resistance, human Strength only;
#   infernal_regalia: Fire Resistance for both;
#   draconic_scale: frost dragonborn takes no damage from /damage ... freeze (health stays 20), human takes it.
wait 40
closescreen
/gamemode survival
/tp @s 0.5 150 0.5 0 0
/effect give @s minecraft:slow_falling 600 0 true
/clear @s
/dndrace set @s elf
wait 10
closescreen
/effect clear @s
/effect give @s minecraft:slow_falling 600 0 true
/item replace entity @s armor.head with dndclasses:elven_chain_helmet
/item replace entity @s armor.chest with dndclasses:elven_chain_chestplate
/item replace entity @s armor.legs with dndclasses:elven_chain_leggings
/item replace entity @s armor.feet with dndclasses:elven_chain_boots
wait 20
/say SET elven_chain RACE elf
/data get entity @s ActiveEffects
/data get entity @s Health
/effect give @s minecraft:instant_health 1 3 true
/dndrace set @s human
