# Portable Hole (docs/items/magic-items.md#portable-hole). Run:
#   timeout 400 ./gradlew runClient -PdevScript=devscripts/portable-hole.txt
# then portable-hole-relog.txt (the contents survive the relog).
# Expect in latest.log:
#  1. storage: after quick-moving 5 diamonds in, `slots` lists them in hole slot 0; after dropping the cloth and picking
#     it up again, they're still there ("[Magic] Dev closed a Portable Hole (1 stacks inside)")
#  2. passage: sneak + use on a 3-thick stone wall: "[Magic] Dev laid a Portable Hole ... 6 blocks, 3 deep, through",
#     "Test passed" for the passage block at 3000 150 3004, then (waitserver 300) "Test passed" for stone again
#  3. a chest in the wall: "couldn't lay a Portable Hole ... (blocked)", the stone behind it untouched
#  4. right-clicking an open passage folds it up early: "passage closed (Dev (right-click))"
#  5. the Bag in the Hole: "Astral rift ... (BAG_IN_HOLE)", the husk's Health 12.0, no Portable Hole left (/clear ... 0)
wait 40
closescreen
include _reset.txt
/difficulty easy
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/gamerule keepInventory true
/time set noon
/kill @e[tag=holetest]
/gamemode creative
/tp @s 3000.5 152 3000.5 0 0
wait 40
/fill 2990 149 2990 3010 149 3015 minecraft:stone
/fill 2990 150 2990 3010 156 3015 minecraft:air
/gamemode survival
/clear @s
/effect give @s minecraft:resistance 600 4 true
/dndmagic give @s dndclasses:portable_hole
/item replace entity @s hotbar.1 with minecraft:diamond 5
/tp @s 3000.5 150 3000.5 0 -40
hotbar 0
wait 10
# 1. Storage: open, quick-move the diamonds in, look
use
wait 10
slots
slot 82 quick_move
wait 5
slots
closescreen
wait 5
# Drop the cloth and pick it up again
press key.drop
wait 50
/tp @s @e[type=item,limit=1,sort=nearest]
wait 20
/tp @s 3000.5 150 3000.5 0 -40
hotbar 0
wait 45
use
wait 10
slots
closescreen
wait 45
# 2. A passage through a 3-thick wall
/fill 2998 150 3003 3002 153 3005 minecraft:stone
/tp @s 3000.5 150 3001.5 0 10
wait 10
sneak on
wait 3
use
wait 5
sneak off
actionbar
/execute if block 3000 150 3004 dndclasses:portable_hole_passage
/execute if block 3000 151 3005 dndclasses:portable_hole_passage
/execute if block 3000 152 3004 minecraft:stone
screenshot portable-hole-passage
/tp @s 3000.5 150 3006.5 180 10
wait 5
screenshot portable-hole-through
waitserver 300
/execute if block 3000 150 3004 minecraft:stone
/execute if block 3000 151 3005 minecraft:stone
/tp @s 3000.5 150 3001.5 0 10
wait 45
# 3. A chest in the wall stops it
/setblock 3000 151 3003 minecraft:chest[facing=north]
/setblock 3000 150 3003 minecraft:chest[facing=north]
sneak on
wait 3
use
wait 5
sneak off
actionbar
/execute if block 3000 150 3004 minecraft:stone
/setblock 3000 151 3003 minecraft:stone
/setblock 3000 150 3003 minecraft:stone
wait 45
# 4. Right-clicking the passage folds it up early
sneak on
wait 3
use
wait 5
sneak off
wait 10
/execute if block 3000 150 3003 dndclasses:portable_hole_passage
use
wait 10
/execute if block 3000 150 3003 minecraft:stone
closescreen
wait 45
# 5. The Bag in the Hole: an Astral rift
/summon minecraft:husk 3002.5 150 3000.5 {NoAI:1b,Tags:["holetest"],PersistenceRequired:1b}
/item replace entity @s hotbar.2 with dndclasses:bag_of_holding
/tp @s 3000.5 150 3000.5 0 -40
hotbar 0
wait 10
use
wait 10
slot 83 quick_move
wait 10
/data get entity @e[tag=holetest,limit=1] Health
/clear @s dndclasses:portable_hole 0
/clear @s dndclasses:bag_of_holding 0
screenshot portable-hole-rift
/kill @e[tag=holetest]
/effect clear @s
wait 10
quit
