# Checks attunement (docs/systems/magic-items.md). Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/attunement.txt
# Expect in latest.log ([CHAT] lines), in order:
#  1. an unbonded Cloak of Protection: armor and toughness unchanged
#  2. Items tab Attune: "You attune to [Cloak of Protection]. (1/3 bonds)" after the 3 s channel; armor +1, toughness +1
#  3. bonds 3/3, then a 4th attune (command and tab) refused: "You can only be attuned to 3 items."
#  4. anti-magic: armor back to base while it lasts, +1 again after
#  5. channel cancelled: "Attunement interrupted: you left the table."
#  6. a lost item shows "missing" with Release; Release: "You end your bond with"
#  7. death (keepInventory): the bond is still listed and armor is still +1 after respawn
#  8. short rest: "While you rest, you study your gear." and "You attune to"
#  9. Forge Cleric: Bless +1 on the held iron sword: attack damage +1, "+1 Iron Sword"; a long rest ends it
# 10. Thief Rogue: a Staff of Ice is active=true; as a Fighter active=false
# 11. Artificer level 5: 4 slots ("/4 items")
# Screenshots: attune-tab-before, attune-channel, attune-tab-after, attune-full, attune-missing, attune-forge, attune-tooltip
wait 40
closescreen
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/gamerule keepInventory true
/time set noon
/fill 0 149 0 30 149 8 minecraft:stone
/fill 0 150 0 30 154 8 minecraft:air
/dndclass set @s fighter
wait 10
closescreen
/clear @s
/effect clear @s
/dndmagic release @s all
/setblock 4 150 4 dndclasses:attunement_table
/tp @s 4.5 150 1.5 0 30
hotbar 0
# 1. Unbonded
/attribute @s minecraft:generic.armor get
/attribute @s minecraft:generic.armor_toughness get
/dndmagic give @s dndclasses:cloak_of_protection
wait 30
/say step1-unbonded
/attribute @s minecraft:generic.armor get
/attribute @s minecraft:generic.armor_toughness get
# 2. Attune at the table
use
wait 10
click -84 -104
wait 5
screenshot attune-tab-before
click 95 -21
wait 20
screenshot attune-channel
wait 50
screenshot attune-tab-after
closescreen
wait 30
/say step2-bonded
/attribute @s minecraft:generic.armor get
/attribute @s minecraft:generic.armor_toughness get
/dndmagic info @s
# 3. Cap
/dndmagic give @s dndclasses:cloak_of_protection
/dndmagic give @s dndclasses:cloak_of_protection
/dndmagic give @s dndclasses:cloak_of_protection
hotbar 1
wait 2
/dndmagic attune @s
hotbar 2
wait 2
/dndmagic attune @s
hotbar 3
wait 2
/say step3-fourth
/dndmagic attune @s
/dndmagic bonds @s
use
wait 10
click -84 -104
wait 5
click 95 33
wait 10
screenshot attune-full
closescreen
# 4. Anti-magic
/say step4-antimagic
/effect give @s dndclasses:anti_magic 4
wait 25
/attribute @s minecraft:generic.armor get
wait 100
/attribute @s minecraft:generic.armor get
# 5. Channel cancelled
/dndmagic release @s all
hotbar 0
use
wait 10
click -84 -104
wait 5
click 95 -21
wait 15
/tp @s 25.5 150 4.5
wait 10
/say step5-cancel
/dndmagic bonds @s
# 6. Missing + Release
/tp @s 4.5 150 1.5 0 30
/dndmagic attune @s
/item replace entity @s hotbar.0 with minecraft:air
wait 5
use
wait 10
click -84 -104
wait 5
screenshot attune-missing
/say step6-release
click 95 -77
wait 10
closescreen
/dndmagic bonds @s
# 7. Death
/clear @s
/dndmagic give @s dndclasses:cloak_of_protection
hotbar 0
wait 2
/dndmagic attune @s
/kill @s
wait 20
respawn
wait 40
/say step7-death
/dndmagic bonds @s
/attribute @s minecraft:generic.armor get
# 8. Short rest
/dndmagic release @s all
/tp @s 4.5 150 1.5 0 30
wait 5
/say step8-rest
/dndclass rest @s short
/dndmagic bonds @s
# 9. Forge Cleric
/dndclass set @s cleric
wait 10
closescreen
/dndclass xp @s set 250
/dndclass subclass @s cleric.forge
/dndclass rest @s long
/clear @s
/give @s minecraft:iron_sword
hotbar 0
wait 5
/attribute @s minecraft:generic.attack_damage get
use
wait 10
click -84 -104
wait 5
screenshot attune-forge
/say step9-bless
click 95 -21
wait 10
closescreen
wait 5
/attribute @s minecraft:generic.attack_damage get
/dndmagic info @s
tooltips 0
wait 5
screenshot attune-tooltip
closescreen
/dndclass rest @s long
wait 5
/attribute @s minecraft:generic.attack_damage get
/dndmagic info @s
# 10. Thief
/dndclass set @s rogue
wait 10
closescreen
/dndclass xp @s set 250
/dndclass subclass @s rogue.thief
/clear @s
/give @s dndclasses:staff_of_ice
wait 2
/say step10-thief
/dndmagic info @s
/dndclass set @s fighter
wait 10
closescreen
/dndmagic info @s
# 11. Artificer
/dndclass set @s artificer
wait 10
closescreen
/dndclass xp @s set 700
/say step11-artificer
/dndmagic bonds @s
# Leave a bonded cloak for attunement-relog.txt
/dndclass set @s fighter
wait 10
closescreen
/clear @s
/dndmagic release @s all
/dndmagic give @s dndclasses:cloak_of_protection
hotbar 0
wait 2
/dndmagic attune @s
/say done
wait 20
quit
