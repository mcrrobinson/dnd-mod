# A fresh Goblin Warren made with /place structure in new chunks. Shoots the cave mouth (knoll, timber
# frame, palisade, totems) from outside, then cuts it away and shoots the layout from overhead.
# 2448, -2416 is about 3400 blocks out: Challenge III (usually).
# Destructive (/dungeon cutaway): use a scratch copy of the run dir.
# Run: timeout 500 ./gradlew runClient -PdevRunDir=run-scratch -PdevScript=devscripts/dungeon-warren.txt
# If /place says "Failed to place structure", the ground there failed the site check (or the biome isn't a
# warren biome): try another spot.
# Check: screenshots/dungeon-warren-*.png; the "/dungeon info" [CHAT] lines in logs/latest.log.
wait 40
closescreen
include _reset.txt
/time set noon
/weather clear
/gamerule doDaylightCycle false
/gamemode creative
/tp @s 2455 200 -2409
wait 200
/place structure dndclasses:goblin_warren 2448 64 -2416
wait 100
/dungeon info
# The mouth: stand on the ground by the entrance, then back off and look at the knoll
/spreadplayers 2455 -2409 0 1 false @s
wait 40
/tp @s ~-16 ~6 ~-16 -45 20
wait 60
hud off
screenshot dungeon-warren-mouth
hud on
/tp @s ~32 ~ ~ 45 20
wait 40
hud off
screenshot dungeon-warren-mouth-2
hud on
# Overhead, with the dungeon cut away
/gamemode spectator
/dungeon cutaway
wait 60
/tp @s 2455 140 -2409
look 0 90
wait 200
hud off
screenshot dungeon-warren-overhead
hud on
quit
