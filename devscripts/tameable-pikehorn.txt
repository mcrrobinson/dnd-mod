# Tames a River Pikehorn with raw cod, toggles sitting, then hits a zombie and checks the Pikehorn joins in.
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/tameable-pikehorn.txt
# Output: run/screenshots/pikehorn-*.png. In latest.log: the Owner line after taming, Sitting 1b/0b/1b/0b
# from the sit toggles, the zombie's Health dropping from 19 (one punch) as the Pikehorn bites and burns it,
# and the last zombie's Health below 20 (a wild Pikehorn fighting back).
/time set day
/weather clear
/gamemode survival
/kill @e[type=dndclasses:river_pikehorn]
/kill @e[type=minecraft:zombie]
/fill -10 149 -10 10 149 10 glass
/fill -10 150 -10 10 158 10 air
/effect give @s resistance 120 4 true
/effect give @s fire_resistance 120 0 true
/clear @s
wait 40
closescreen

# Taming: feed cod until it's tamed (1 in 3 per fish). Slowness keeps it in front of the crosshair.
/summon dndclasses:river_pikehorn 0 150 0 {Rotation:[0f,0f],ActiveEffects:[{Id:2,Amplifier:10b,Duration:2000,ShowParticles:0b}]}
/give @s minecraft:cod 32
hotbar 0
/tp @s 0 150 2.2 180 35
wait 20
/data get entity @e[type=dndclasses:river_pikehorn,limit=1] Owner

# Sitting: tamed ones start sitting; empty-hand right-clicks toggle it
hotbar 5
/data get entity @e[type=dndclasses:river_pikehorn,limit=1] Sitting
use
wait 10
/data get entity @e[type=dndclasses:river_pikehorn,limit=1] Sitting
use
wait 30
/data get entity @e[type=dndclasses:river_pikehorn,limit=1] Sitting
/tp @s 2 151 2.5 135 25
wait 5
screenshot pikehorn-sitting
/tp @s 0 150 2.2 180 35
wait 5
use
wait 10
/data get entity @e[type=dndclasses:river_pikehorn,limit=1] Sitting

# Fighting: punch a zombie; the Pikehorn attacks what its owner attacks
/effect clear @e[type=dndclasses:river_pikehorn]
/summon minecraft:zombie 3 150 -1 {Tags:["pk_target"],NoAI:1b,PersistenceRequired:1b,Rotation:[0f,0f],ArmorItems:[{},{},{},{id:"minecraft:leather_helmet",Count:1b}]}
/tp @s 3 150 1.3 180 20
wait 10
attack
wait 5
/data get entity @e[type=minecraft:zombie,tag=pk_target,limit=1] Health
/tp @s 7 151.5 2 120 20
wait 30
screenshot pikehorn-fight
wait 40
/data get entity @e[type=minecraft:zombie,tag=pk_target,limit=1] Health
wait 60
/data get entity @e[type=minecraft:zombie,tag=pk_target,limit=1] Health
screenshot pikehorn-fight2
/execute unless entity @e[type=minecraft:zombie,tag=pk_target]

# Wild ones fight back: a zombie hits a wild Pikehorn and its health should drop
/kill @e[type=dndclasses:river_pikehorn]
/kill @e[type=minecraft:zombie,tag=pk_target]
/summon dndclasses:river_pikehorn -4 150 0 {Tags:["pk_wild"],Rotation:[180f,0f]}
/summon minecraft:zombie -4 150 -2 {Tags:["pk_wild_target"],NoAI:1b,PersistenceRequired:1b,ArmorItems:[{},{},{},{id:"minecraft:leather_helmet",Count:1b}]}
wait 5
/damage @e[tag=pk_wild,limit=1] 1 minecraft:mob_attack by @e[tag=pk_wild_target,limit=1]
wait 100
/data get entity @e[tag=pk_wild_target,limit=1] Health
/kill @e[type=dndclasses:river_pikehorn]
/kill @e[type=minecraft:zombie]
/gamemode creative
wait 10
quit
