# Setting the class to none reopens the class picker; a classed player who dies keeps the class. Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/class-none-fallthrough.txt
# Expect screenshots: class-none-picker shows the picker, class-none-after-death doesn't.
wait 40
closescreen
wait 10
/dndclass set @s wizard
wait 20
/dndclass set @s none
wait 20
screenshot class-none-picker
closescreen
wait 10
/dndclass set @s fighter
wait 20
/kill @s
wait 40
respawn
wait 40
screenshot class-none-after-death
closescreen
wait 10
/dndclass set @s barbarian
wait 20
quit
