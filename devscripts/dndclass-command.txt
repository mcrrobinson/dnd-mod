# The /dndclass set|get command: class stats change with the class, unknown classes are rejected, none opens the picker. Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/dndclass-command.txt
# Expect in latest.log: max health / speed change between Barbarian and Wizard, an error for notaclass; dndclass-none screenshot shows the picker.
wait 40
closescreen
wait 10
/dndclass set @s barbarian
wait 10
/dndclass get @s
/attribute @s minecraft:generic.max_health get
/attribute @s minecraft:generic.movement_speed base get
wait 10
/dndclass set @s wizard
wait 10
/attribute @s minecraft:generic.max_health get
/attribute @s minecraft:generic.movement_speed base get
/dndclass set @s barbarian
wait 10
/attribute @s minecraft:generic.max_health get
/dndclass set @s notaclass
wait 10
screenshot dndclass-command
/dndclass set @s none
wait 20
screenshot dndclass-none
closescreen
quit
