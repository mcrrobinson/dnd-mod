# Checks the Ancient lair's guard posts raise kobolds now the Kobold Tribes exist (docs/structures/ancient-lairs.md).
# Uses the dev world's lair near 1088 -2492 (ancient-lair-seeds.txt).
# Run: timeout 300 ./gradlew runClient -PdevScript=devscripts/ancient-lair-kobolds.txt
# Output: latest.log has "[LairGuards] a dndclasses:kobold... takes the guard post" lines, two "Test passed" counts and
# "Test failed" for the marker check (no empty posts left);
# run/screenshots/ancient_lair-kobolds.png shows guards on the crater floor.
/time set day
/weather clear
wait 40
closescreen
include _reset.txt
/difficulty normal
/gamemode spectator
hud off
/tp @s 1088 200 -2492
wait 300
/dndhome ancientlair air
wait 300
waitserver 2
/execute if entity @e[tag=dndclasses.lair_guard,distance=..120]
/execute if entity @e[type=#dndclasses:faction/kobolds,tag=dndclasses.lair_guard,distance=..120]
/execute if entity @e[type=marker,tag=dndclasses.lair_guard_post,distance=..120]
/tp @s @e[type=#dndclasses:faction/kobolds,tag=dndclasses.lair_guard,limit=1,sort=nearest]
/tp @s ~ ~4 ~-6 0 35
wait 60
screenshot ancient_lair-kobolds
quit
