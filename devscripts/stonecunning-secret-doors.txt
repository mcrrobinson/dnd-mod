# Stonecunning, part 1: places a Dwarven Ruin (as dungeon-ruin.txt does) and lists its hidden stone doorways.
# Places a structure: use a scratch copy of the run dir.
# Run: timeout 500 ./gradlew runClient --offline -PdevRunDir=run-prt -PdevScript=devscripts/stonecunning-secret-doors.txt
# Check: the "/dndrace traits" [CHAT] line lists "hidden stone doorways within 128 blocks: x y z (9 blocks)".
# Part 2 (stonecunning-secret-doors-2.txt) stands a sneaking Dwarf next to one of them.
wait 40
closescreen
/time set noon
/weather clear
/gamerule doDaylightCycle false
/gamemode creative
/dndrace set @s dwarf
/tp @s 2455 200 -2409
/forceload add 2368 -2496 2528 -2336
wait 400
/place structure dndclasses:dwarven_ruin 2448 64 -2416
wait 100
/dungeon info
/spreadplayers 2455 -2409 0 1 false @s
wait 40
/dndrace traits @s
/forceload remove all
quit
