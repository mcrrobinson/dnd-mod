# Reputation tiers applied to NPCs. Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/rep-tiers.txt
# Expect in the log ([DevScript]/[CHAT]/[D20]) and screenshots (rep-*.png):
# - innkeeper trades: Unfriendly cake 5 emeralds (base 3), wheat 30 (base 20); Exalted cake 2, wheat 12; Hostile: no screen
# - bounty board refuses a Hostile player (rep-board-hostile)
# - hobbit: Hostile flees out of 5 blocks, no gift; gift cooldown 18000 at Unfriendly, 3000 at Exalted
# - dwarf: Neutral no target, Hostile targets the player; hoard witness at 20 blocks only when Unfriendly
# - goblin: Neutral untouched for 5 s next to it, Unfriendly ignores at 10 blocks but targets at 4, Hostile targets
# - Persuasion DC 12 +3 at Unfriendly (15) and -6 at Exalted (6), with the test datapack faction dndtest:villagers
wait 40
closescreen
/difficulty normal
/gamemode survival
# Not a Bard: hostile mobs ignore Bards, which would hide the goblin results.
/dndclass set @s fighter
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/time set noon
/effect give @s minecraft:resistance 900 4 true
/effect give @s minecraft:regeneration 900 4 true
/kill @e[type=!minecraft:player,distance=..60]
/fill -12 149 -12 24 149 24 minecraft:stone
/fill -12 150 -12 24 154 24 minecraft:air
/clear @s
/tp @s 0.5 150 0.5 0 35
wait 40

# ---- Innkeeper prices ----
/summon dndclasses:innkeeper 0.5 150 2.5 {NoAI:1b,Rotation:[180f,0f],Tags:["rt"]}

wait 5

screenshot rep-innkeeper-exalted

wait 5
/rep @s dndclasses:hobbits set -600
wait 5
use
wait 3

slots

/kill @e[tag=rt]

/kill @e[type=minecraft:item]
/clear @s
/summon dndclasses:hobbit 0.5 150 2.5 {NoAI:1b,Rotation:[180f,0f],Tags:["rh"]}
wait 5
use
wait 3
screenshot rep-hobbit-hostile-gift
/data get entity @e[tag=rh,limit=1] GiftCooldown
/rep @s dndclasses:hobbits set -200
/clear @s
wait 3
use
wait 3
/data get entity @e[tag=rh,limit=1] GiftCooldown
/data merge entity @e[tag=rh,limit=1] {GiftCooldown:0}
/rep @s dndclasses:hobbits set 800
/clear @s
wait 3
use
wait 3
/data get entity @e[tag=rh,limit=1] GiftCooldown
/kill @e[tag=rh]
/clear @s
/rep @s dndclasses:hobbits set 0
