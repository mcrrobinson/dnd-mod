# Hireling role abilities and party synergies (docs/systems/hirelings.md "Role abilities",
# docs/systems/party-synergies.md "Hirelings"). Solo player, high above the dev world at x 40-70, z 40-70.
# Run: timeout 500 ./gradlew runClient --no-daemon --offline -PdevScript=devscripts/hireling-roles.txt
# Expect in logs/latest.log:
#  1. Healer: "[Party] Triage: Mira (hireling) heals Dev for 7.5 instead of 6.0" and
#     "[Hireling] Mira Cure Wounds -> Dev (7.5 HP, ends poisoned)"; /dndcondition list no longer shows poisoned.
#  2. Revivify: Downed with a fresh Healer: "[Hireling] Sorrel stabilises Dev (20 ticks)",
#     "[Hireling] Sorrel Revivify -> Dev (6.0 HP)", "[Downed] Dev stood up with 6.0 HP".
#  3. Tank + Damage hirelings, 3 zombies: "[Hireling] Brannoc Taunt -> Zombie (was on Dev)" for each, the
#     `target` readouts name Brannoc's hireling, "[Hireling] Grisha Hunter's Mark -> Zombie (+6.0, it was on
#     Brannoc)" and "[Party] Opportunist: Grisha (hireling) hits Zombie (on Brannoc)".
#  4. Balanced party: a solo Wizard (Damage) with Tank and Healer hirelings: "[Party] Balanced party: ... with 3
#     role(s) (hirelings: [...]): x1.1".
#  5. Support: "[Hireling] Lyra Rallying Song -> Dev (no longer frightened), Lyra (Regeneration I, 100 ticks)";
#     a Fighter's obstacle check: "[Party] Inspire: Dev rolls dndclasses:collapsed_passage with advantage
#     (Lyra (hireling))".
#  6. Utility: "[Perception] Tamsin (hireling) spots obstacle ... for Dev (passive 15 vs DC 13) -> revealed" and
#     "[Obstacle] Dev found the dndclasses:hidden_door"; in a dark tunnel "[Hireling] Tamsin Torchbearer -> torch
#     at ...", and Torches drops below 16.
wait 40
closescreen
include _reset.txt
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/gamerule dndDeathSaves 1
/gamerule dndLastStand true
/gamerule sendCommandFeedback true
/time set noon
/weather clear
/party leave
/hireling dismiss all
/fill 40 149 40 70 149 70 minecraft:stone
/fill 40 150 40 70 156 70 minecraft:air
/tp @s 45.5 150 45.5 0 10
wait 20
/kill @e[type=dndclasses:hireling,distance=..60]
/kill @e[tag=hrtest]
/dndclass set @s wizard
/clear @s
/effect give @s minecraft:instant_health 1 5
# A Wizard has 10 max health: the damage below is sized for that.
/attribute @s minecraft:generic.max_health get

# ---- 1. Healer: Cure Wounds (Triage x1.25 on a player) ends Poisoned
/hireling admin spawn @s healer human Mira
wait 10
/dndcondition apply @s poisoned 400
/damage @s 6 minecraft:generic
wait 40
/dndcondition list @s
/data get entity @s Health
screenshot hireling-roles-cure

# ---- 2. Revivify: a Healer stabilises in 20 ticks and stands you up with 6 HP
/hireling dismiss all
wait 5
/kill @e[type=dndclasses:hireling,distance=..60]
/effect give @s minecraft:instant_health 1 5
/hireling admin spawn @s healer elf Sorrel
/tp @e[type=dndclasses:hireling,limit=1,sort=nearest] 45.5 150 48.5
wait 20
/damage @s 12 minecraft:generic
wait 10
screenshot hireling-roles-revivify
wait 50
/data get entity @s Health
/dndclass revive @s

# ---- 3. Tank Taunt, Damage Hunter's Mark and Opportunist
/hireling dismiss all
wait 5
/kill @e[type=dndclasses:hireling,distance=..60]
/effect give @s minecraft:instant_health 1 5
/effect give @s minecraft:resistance 120 4
/tp @s 45.5 150 45.5 0 10
/hireling admin spawn @s tank human Brannoc
/hireling admin spawn @s damage half_orc Grisha
/tp @e[type=dndclasses:hireling,nbt={Role:"tank"}] 45.5 150 48.5
/tp @e[type=dndclasses:hireling,nbt={Role:"damage"}] 41.5 150 44.5
wait 10
/summon minecraft:zombie 44.5 150 52.5 {Tags:["hrtest"],ArmorItems:[{},{},{},{id:"minecraft:iron_helmet",Count:1b}]}
/summon minecraft:zombie 45.5 150 52.5 {Tags:["hrtest"],ArmorItems:[{},{},{},{id:"minecraft:iron_helmet",Count:1b}]}
/summon minecraft:zombie 46.5 150 52.5 {Tags:["hrtest"],ArmorItems:[{},{},{},{id:"minecraft:iron_helmet",Count:1b}]}
wait 25
target @e[tag=hrtest]
screenshot hireling-roles-taunt
wait 20
target @e[tag=hrtest]
wait 80
/kill @e[tag=hrtest]

# ---- 4. Balanced party: Wizard (Damage) + Tank + Healer hirelings = 3 roles, +10% class XP
/hireling dismiss Grisha
wait 5
/kill @e[type=dndclasses:hireling,nbt={Role:"damage"},distance=..60]
/hireling admin spawn @s healer gnome Pip
wait 10
/summon minecraft:zombie 48.5 150 48.5 {Tags:["hrtest","hrxp"],NoAI:1b}
/summon minecraft:zombie 49.5 150 48.5 {Tags:["hrtest","hrxp"],NoAI:1b}
/summon minecraft:zombie 50.5 150 48.5 {Tags:["hrtest","hrxp"],NoAI:1b}
/summon minecraft:zombie 51.5 150 48.5 {Tags:["hrtest","hrxp"],NoAI:1b}
/summon minecraft:zombie 52.5 150 48.5 {Tags:["hrtest","hrxp"],NoAI:1b}
wait 5
/execute as @e[tag=hrxp] run damage @s 1000 minecraft:player_attack by @p
wait 10
/dndclass progress @s
/kill @e[tag=hrtest]

# ---- 5. Support: Rallying Song (hurt and Frightened) and Inspire on an obstacle check
/hireling dismiss all
wait 5
/kill @e[type=dndclasses:hireling,distance=..60]
/effect clear @s
/effect give @s minecraft:instant_health 1 5
/hireling admin spawn @s support tiefling Lyra
wait 10
/dndcondition apply @s frightened 400
/damage @s 6 minecraft:generic
wait 30
/dndcondition list @s
screenshot hireling-roles-song
/dndclass set @s fighter
/clear @s
/dndobstacle place 44 150 58 46 151 58 dndclasses:collapsed_passage medium
/tp @s 45.5 150 56.5 0 20
/tp @e[type=dndclasses:hireling,limit=1,sort=nearest] 47.5 150 55.5
wait 10
/dndclass forceroll @s 15
use
wait 40
/execute if block 45 150 58 dndclasses:collapsed_passage[state=open]

# ---- 6. Utility: Scout reveals a hidden door; Torchbearer lights a dark tunnel
/hireling dismiss all
wait 5
/kill @e[type=dndclasses:hireling,distance=..60]
/dndclass set @s wizard
/clear @s
/fill 40 150 64 50 153 64 minecraft:stone_bricks
/dndobstacle place 45 150 64 45 151 64 dndclasses:hidden_door medium
/tp @s 45.5 150 59.5 0 10
/hireling admin spawn @s utility dragonborn Tamsin
/tp @e[type=dndclasses:hireling,limit=1,sort=nearest] 46.5 150 61.5
wait 40
/execute if block 45 150 64 dndclasses:hidden_door[state=sealed]
screenshot hireling-roles-scout
/fill 57 149 40 62 153 70 minecraft:stone hollow
/tp @s 59.5 150 42.5 0 10
/tp @e[type=dndclasses:hireling,limit=1,sort=nearest] 59.5 150 41.5
wait 20
/tp @s 59.5 150 45.5 0 10
wait 15
/tp @s 59.5 150 48.5 0 10
wait 15
/tp @s 59.5 150 51.5 0 10
wait 15
/tp @s 59.5 150 54.5 0 10
wait 15
/tp @s 59.5 150 57.5 0 10
wait 15
/tp @s 59.5 150 60.5 0 10
wait 15
/tp @s 59.5 150 63.5 0 10
wait 15
/tp @s 59.5 150 66.5 0 10
wait 60
/data get entity @e[type=dndclasses:hireling,limit=1,sort=nearest] Torches
/tp @s 59.5 150 66.5 180 20
wait 5
screenshot hireling-roles-torches
/fill 57 149 40 62 153 70 minecraft:air
/hireling dismiss all
wait 5
/kill @e[type=dndclasses:hireling,distance=..60]
quit
