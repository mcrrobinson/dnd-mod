# Visits a Nether Fortress generated after the warlord was added and checks it has its Goblin Warlord.
# Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/goblin-warlord-fortress.txt
# The coordinates are from "/locate structure minecraft:fortress" at 30000 90 30000 in the dev world; pick fresh
# ones if those chunks were already generated before the warlord existed.
# Expect in latest.log: "Test passed, count: 1", the warlord's Pos near its GuardPos, PersistenceRequired 1b.
wait 40
closescreen
/gamemode creative
/execute in minecraft:the_nether run tp @s 30121 90 29929
wait 300
/execute if entity @e[type=dndclasses:goblin_warlord]
/execute as @e[type=dndclasses:goblin_warlord] run data get entity @s Pos
/execute as @e[type=dndclasses:goblin_warlord] run data get entity @s GuardPos
/execute as @e[type=dndclasses:goblin_warlord] run data get entity @s PersistenceRequired
/execute as @e[type=dndclasses:goblin_warlord] at @s run tp @p ~5 ~2.5 ~5 135 20
wait 60
screenshot warlord-fortress
/execute in minecraft:overworld run tp @s 0 151 0
wait 10
quit
