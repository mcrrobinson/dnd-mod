wait 40
closescreen
wait 10
/gamemode creative
/tp @s 30 160 25 180 20
wait 20
/fill 20 140 -5 40 160 15 minecraft:air

wait 20
/data get block 30 150 5 BrewTime
wait 400
/execute if block 30 150 5 minecraft:brewing_stand
/execute if block 30 149 5 minecraft:stone
wait 200
/execute if block 30 150 5 minecraft:brewing_stand

screenshot brewing-explosion
