# Verify PR #214 (camp goblins keep their leash). Run after goblin-camp-leash.txt in the same run dir, as a new
# launch, so the first readout is after a quit and rejoin. Camp nearest 40000 40000: fire at 38312 77 39032.
# Run: timeout 900 ./gradlew runClient -PdevScript=devscripts/goblin-camp-leash-verify.txt
# Check latest.log after each "camp-<step>" chat line: warrior Pos lines (horizontal distance from 38312 39032),
# CampCenter lines, and "[DevScript] target" lines.
#   camp-rejoin: every warrior within about 12 of the fire, CampCenter 38312 77 39032.
#   camp-wander-*: the warriors moved ~28 out walk back within about 12; the leash-less persistent stand-in
#     (tag oldcamp) picks up CampCenter and walks in too.
#   camp-chase: warriors (tag campg) target Dev; camp-lost-*: back within about 12. The camp's tents and palisade
#     often stop them following far: goblin-leash-chase.txt tests a chase past the leash on open ground.
#   camp-others: non-camp goblins (leftover raiders, waves): "Test failed" = none has CampCenter or is persistent.
#   camp-raid-*: "Test failed" (no raider has CampCenter); raiders march to the rally (38352 39032) and target Dev.
#   camp-wave-*: summoned on uneven ground the wave can fail to place; goblin-camp-chase-wave.txt uses a platform.
wait 40
closescreen
include _reset.txt
wait 40
closescreen
/dm off
/gamemode spectator
/gamerule doDaylightCycle false
/time set 6000
/weather clear
/tp @s 38312 160 39032 0 90
wait 300
waitserver 2
/say camp-rejoin
/execute as @e[type=dndclasses:goblin_warrior,x=38232,y=-64,z=38952,dx=160,dy=400,dz=160] run data get entity @s Pos
/tag @e[type=dndclasses:goblin_warrior,x=38232,y=-64,z=38952,dx=160,dy=400,dz=160,nbt={CampCenter:{}}] add campg
/execute as @e[type=dndclasses:goblin_warrior,x=38232,y=-64,z=38952,dx=160,dy=400,dz=160] run data get entity @s CampCenter
wait 600
/say camp-rejoin-30s
/execute as @e[type=dndclasses:goblin_warrior,x=38232,y=-64,z=38952,dx=160,dy=400,dz=160] run data get entity @s Pos
