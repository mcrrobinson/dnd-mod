# Craft and element obstacles (Artificer, Rogue, Alchemist, Warlock, and a Fighter's fallbacks).
# Run: timeout 500 ./gradlew runClient -PdevScript=devscripts/obstacle-craft-element.txt
# Expect (log: [DevScript] actionbar lines, [CHAT] command feedback, [Obstacle]/[Grudges] lines; screenshots):
#  Clockwork Door: an Artificer with no parts is refused ("You need an iron ingot..."); with them, a rigged 18 opens
#   it and uses up one iron ingot and one redstone dust (1 of each left). A rigged 1 jams it: the next try says
#   "jammed" and spends nothing. A Rogue (secondary) opens another one.
#  Smashed Clockwork Door: a Fighter mines it open; the watching Mountain Dwarf reacts ([Grudges] ... BREAK).
#  Alchemical Seal: its chest is glued shut; an Alchemist with no potion is refused; a rigged 18 dissolves it and
#   uses up the potion; a rigged 1 splashes acid (health down, Poison). A Fighter breaks a second seal: its chest
#   rolls its loot and some stacks are ruined ([Obstacle] ... stack(s) ruined).
#  Miasma Vent: a Fighter by it gets Poison II (Amplifier 1) and Nausea; a Rogue gets nothing; an Alchemist's
#   rigged 18 stops it, and the Fighter then gets nothing.
#  Fire Curtain: a Fighter walking in is stopped before it (Pos z < 30) and scorched; a drunk Potion of Fire
#   Resistance doesn't take (no ActiveEffects) and he's still stopped. A Warlock walks through (z > 32) unhurt and
#   douses the brazier: the curtain opens. After a reset, an Alchemist's Transmute of Fire Resistance covers the
#   Fighter (ActiveEffects has id 12) and he walks through.
wait 40
closescreen
include _reset.txt
respawn
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/gamerule doImmediateRespawn true
/time set noon
/kill @e[type=!minecraft:player]
/fill 0 149 0 14 149 44 minecraft:stone
/fill 0 150 0 14 156 44 minecraft:air

# ---- Artificer: Clockwork Door (Medium)
/dndobstacle place 3 150 5 4 151 5 dndclasses:clockwork_door medium
/dndclass set @s artificer
wait 5
closescreen
/clear @s
/effect clear @s
/tp @s 3.5 150 3.5 0 20
wait 10
screenshot craft-clockwork-hint
use
wait 2
actionbar
/execute if block 3 150 5 dndclasses:clockwork_door[state=sealed]
/give @s minecraft:iron_ingot 2
/give @s minecraft:redstone 2
/dndclass forceroll @s 18
use
wait 40
/execute if block 3 150 5 dndclasses:clockwork_door[state=open]
/clear @s minecraft:iron_ingot 0
/clear @s minecraft:redstone 0
screenshot craft-clockwork-open
# A natural 1 jams it; the next try is refused and spends nothing
/dndobstacle reset 3 150 5
/dndclass forceroll @s 1
use
wait 110
/dndclass forceroll @s 18
use
wait 2
actionbar
/execute if block 3 150 5 dndclasses:clockwork_door[state=sealed]
/clear @s minecraft:iron_ingot 0
/dndclass forceroll @s clear

# ---- Fighter: smash a Clockwork Door in front of a Mountain Dwarf
/dndobstacle place 3 150 10 4 151 10 dndclasses:clockwork_door medium
/summon dndclasses:mountain_dwarf 8.5 150 8.5 {NoAI:1b,PersistenceRequired:1b,Rotation:[90f,0f]}

wait 5
closescreen
/clear @s
/item replace entity @s weapon.mainhand with minecraft:netherite_pickaxe{Enchantments:[{id:"minecraft:efficiency",lvl:5}]}
/tp @s 3.5 150 8.5 0 20

wait 120
mine off
/execute if block 4 150 10 dndclasses:clockwork_door[state=open]
/data get entity @e[type=dndclasses:mountain_dwarf,limit=1] AngryAt
screenshot craft-clockwork-smashed
/kill @e[type=dndclasses:mountain_dwarf]
