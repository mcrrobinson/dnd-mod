# Revives and self-heals on one client (no ally needed: /dndclass down downs anyway).
#  1 a splash Healing potion revives a Rogue
#  2 ...but not a Paladin (potion immunity)
#  3 natural regen (full food + Saturation) doesn't revive; Fighter Second Wind doesn't either
#  4 Regeneration put on after going down (a Druid's Regrowth, a beacon) revives
#  5 Fighter Indomitable rerolls one failed death save per downing
# Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/downed-revive-solo.txt -PdevRunDir=run-3
wait 40
closescreen
respawn
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/gamerule naturalRegeneration true
/gamerule dndDeathSaves 1
/time set noon
/kill @e[type=!player]
/party leave
/fill 0 149 0 20 149 20 minecraft:stone
/fill 0 150 0 20 155 20 minecraft:air
/tp @s 10.5 150 10.5 0 10
/dndclass set @s rogue
wait 10
closescreen
/attribute @s minecraft:generic.max_health base set 20
/effect give @s minecraft:instant_health 1 3
wait 10
# 1: Rogue + splash Healing
/dndclass forceroll @s 12 12
/dndclass down @s
wait 10
/summon minecraft:potion 10.5 151.5 10.5 {Item:{id:"minecraft:splash_potion",Count:1b,tag:{Potion:"minecraft:healing"}}}
wait 20
/dndclass downed @s
/data get entity @s Health
/dndclass forceroll @s clear
wait 30
# 2: Paladin + splash Healing
/dndclass set @s paladin
wait 10
closescreen
/attribute @s minecraft:generic.max_health base set 20
/dndclass forceroll @s 12 12
/dndclass down @s
wait 10
/summon minecraft:potion 10.5 151.5 10.5 {Item:{id:"minecraft:splash_potion",Count:1b,tag:{Potion:"minecraft:healing"}}}
wait 20
/dndclass downed @s
/data get entity @s Health
/dndclass revive @s 20
/dndclass forceroll @s clear
wait 30
# 3: natural regen and Second Wind
/dndclass set @s fighter
wait 10
closescreen
/attribute @s minecraft:generic.max_health base set 20
/dndclass xp @s set 20000
/dndclass unlock @s fighter.second_wind
/dndclass unlock @s fighter.indomitable
/dndclass progress @s
/dndclass forceroll @s 12 12
/dndclass down @s
/effect give @s minecraft:saturation 5 10
wait 100
/dndclass downed @s
/data get entity @s Health
# 4: Regeneration put on now revives
/effect give @s minecraft:regeneration 5 1
wait 30
/dndclass downed @s
/data get entity @s Health
/dndclass forceroll @s clear
wait 30
# 5: Indomitable: the first fail is rerolled (5 -> 15 pass), the second isn't (5 fail)
/effect clear @s
/dndclass forceroll @s 5 15 5 12
/dndclass down @s
wait 125
screenshot revive-indomitable
/dndclass downed @s
wait 120
/dndclass downed @s
/dndclass revive @s 20
/dndclass forceroll @s clear
wait 20
quit
