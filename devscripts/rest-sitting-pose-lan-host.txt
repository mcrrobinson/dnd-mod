# Two clients: the guest sees the host sitting at a campfire during a short rest, then standing again.
#  1. timeout 400 ./gradlew runClient -PdevScript=devscripts/rest-sitting-pose-lan-host.txt
#  2. once its log says "Started serving on":
#     timeout 400 ./gradlew runClient -PdevRunDir=run-2 -PdevServer=localhost:25617 -PdevScript=devscripts/rest-sitting-pose-lan-guest.txt
# Screenshots (in the guest's run dir): lan-guest-sees-sitting, lan-guest-sees-stood.
wait 40
closescreen
include _reset.txt
/gamemode survival
/gamerule doMobSpawning false
/gamerule doDaylightCycle false
/gamerule dndRests true
/time set 1000
/kill @e[type=dndclasses:rest_seat]
/fill -4 149 -4 6 149 6 minecraft:grass_block
/fill -4 150 -4 6 155 6 minecraft:air
/setblock 0 150 2 minecraft:campfire[lit=true]
/tag @s add host
/dndclass set @s fighter
/clear @s
/dndclass rest @s long
/tp @s 0.5 150 0.5 0 35
/publish false survival 25617
waitchat guest-ready
# The guest watches from the side, 3 blocks east
/tp @a[tag=!host,limit=1] 3.5 150 0.5 90 15
/tp @s 0.5 150 0.5 0 35
# (A short rest finished in the last 3 minutes of world time blocks a new one: give the cooldown time)
wait 1600
wait 20
clearchat
sneak on
use
wait 5
sneak off
wait 20
/execute if entity @e[type=dndclasses:rest_seat]
/say host-sitting
waitchat guest-saw-sitting
sneak on
wait 5
sneak off
wait 20
/execute unless entity @e[type=dndclasses:rest_seat]
/say host-stood
waitchat guest-saw-stood
# Let the guest quit first: its script stalls once the host has gone
wait 100
quit
