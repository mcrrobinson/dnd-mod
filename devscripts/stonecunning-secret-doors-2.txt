# Stonecunning, part 2 (after stonecunning-secret-doors.txt, same run dir): a sneaking Dwarf in the ruin's secret
# room sees its bricked-up doorway outlined in gold dust; a Human doesn't.
# Run: timeout 300 ./gradlew runClient --offline -PdevRunDir=run-prt -PdevScript=devscripts/stonecunning-secret-doors-2.txt
# Check: the "/dndrace traits" [CHAT] line lists the doorway; screenshots stonecunning-dwarf (outline) and
# stonecunning-human (none).
wait 40
closescreen
/gamemode creative
/dndrace set @s dwarf
/tp @s 2455.5 -20 -2392.5
wait 100
/dndrace traits @s
# The secret room's way out runs south to the puzzle room, whose north wall hides the doorway (2454-2456 -20..-18
# -2383 for the ruin part 1 places)
/execute if block 2455 -20 -2390 minecraft:air
/execute if block 2455 -20 -2388 minecraft:air
/execute if block 2455 -20 -2386 minecraft:air
/execute if block 2455 -20 -2384 minecraft:air
/execute if block 2455 -20 -2383 minecraft:cracked_deepslate_bricks
/gamemode survival
/effect give @s minecraft:resistance 600 4 true
/tp @s 2455.5 -20 -2388.5 0 5
wait 40
sneak on
wait 40
hud off
screenshot stonecunning-dwarf
hud on
sneak off
wait 10
/dndrace set @s human
sneak on
wait 40
hud off
screenshot stonecunning-human
hud on
sneak off
/dndrace set @s dwarf
/gamemode creative
quit
