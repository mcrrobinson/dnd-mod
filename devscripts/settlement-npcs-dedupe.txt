# The settlement merchants, guards and peaceful NPCs after the merge into NpcMerchant / SettlementNpcEntity /
# GuardAnger (one run before and one after the refactor; the readouts should match).
# Run: timeout 400 ./gradlew runClient -PdevScript=devscripts/settlement-npcs-dedupe.txt
# Expect in <run dir>/logs/latest.log, after each "[nm] ..." /say marker:
# - merchants (innkeeper, Speaker, Fletcher, Elder, Tinker, Quartermaster, Pactbroker): "[DevScript] trade i: ..."
#   lines from slots, with Neutral prices; after buying trade 0 its uses go up by one; with the home race the kin
#   stock is listed after the normal offers at kin prices (the Quartermaster only serves a Half-Orc); the
#   Pactbroker also shows the Wither piece to a Warlock; at Hostile no screen opens and the action bar says
#   "... refuses to trade with you"
# - struck peaceful NPCs (/damage ... by @s): the guard next to them has AngryAt = the player's UUID
#   (wood elf -> Elf Warden, Elder -> Templar, Pactbroker -> Exile and "[BrimstoneSanctum] ... rebuked Dev",
#   Tinker -> Clockwork Sentinel, orc warrior -> orc archer), the hobbit's has none
# - Home / HomeRange / Variant NBT of a Tiefling Exile, Gnome Tinker and wood elf
# - a chest opened next to a watching mountain dwarf: "[Grudges] dndclasses:mountain_dwarves OPEN ... GRUDGE" and
#   its AngryAt
wait 40
closescreen
include _reset.txt
/dm off
/gamemode survival
/gamerule doDaylightCycle false
/gamerule doMobSpawning false
/time set 6000
/effect give @s minecraft:resistance 900 4 true
/kill @e[tag=nm]
/fill 30 149 30 50 149 50 minecraft:stone
/fill 30 150 30 50 156 50 minecraft:air
/clear @s
/tp @s 40.5 150 40.5 0 15
wait 40
/give @s minecraft:emerald 64
hotbar 8
sneak on
wait 3
