# Docs screenshots for class specials whose look is the same on main: Barbarian, Fighter, Artificer, Cleric, Ranger, Blood Hunter.
# Run: timeout 900 ./gradlew runClient -PdevRunDir=run-classes -PdevScript=devscripts/docs-classes-main.txt
# Output: <run dir>/screenshots/cls-*.png. Grass stage at y=200; each class waits ~19 s after the switch so the mana bar is full.
wait 40
closescreen
hud off
/gamerule doDaylightCycle false
/gamerule doWeatherCycle false
/gamerule doMobSpawning false
/gamerule doImmediateRespawn true
/gamerule sendCommandFeedback false
/weather clear
/time set 6000
/gamemode survival
/difficulty normal
/tp @s 1000.5 200 1000.5 0 0
/effect give @s minecraft:slow_falling 30 0 true
wait 60
/fillbiome 980 192 980 1020 207 1020 minecraft:plains
/fill 980 199 980 1020 199 1020 minecraft:grass_block
/kill @e[type=!minecraft:player,distance=..60]
/tp @s 1000.5 200 1000.5 0 0
wait 20
